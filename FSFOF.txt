//Hull Suite

// INPUTS
src = close
string modeSwitch = "Hma"
int length = 55
float lengthMult = 1.0

bool useHtf =false
htf = input.timeframe("240", title="Higher timeframe", group = "Hull Suite")

switchColor = true
candleCol = false
visualSwitch = input.bool(true, title="Show as a Band?", group = "Hull Suite")
thicknessSwitch = input.int(1, title="Line Thickness", group = "Hull Suite")
transpSwitch = input.int(40, title="Band Transparency", step=5, group = "Hull Suite")

// FUNCTIONS
hullMovingAverage(_src, _length) => ta.wma(2 * ta.wma(_src, _length / 2) - ta.wma(_src, _length), math.round(math.sqrt(_length)))
expHullMovingAverage(_src, _length) => ta.ema(2 * ta.ema(_src, _length / 2) - ta.ema(_src, _length), math.round(math.sqrt(_length)))
triHullMovingAverage(_src, _length) => ta.wma(ta.wma(_src, _length / 3) * 3 - ta.wma(_src, _length / 2) - ta.wma(_src, _length), _length)

modeSelection(mode, _src, _len) =>
    if mode == "Hma"
        hullMovingAverage(_src, _len)
    else if mode == "Ehma"
        expHullMovingAverage(_src, _len)
    else if mode == "Thma"
        triHullMovingAverage(_src, _len / 2)
    else
        na

// OUTPUT
_hull = modeSelection(modeSwitch, src, int(length * lengthMult))
HULL = useHtf ? request.security(syminfo.tickerid, htf, _hull) : _hull
MHULL = HULL[0]
SHULL = HULL[2]

// COLOR
hullColor = switchColor ? (HULL > HULL[2] ? color.green : color.red) : color.orange

// PLOT
Fi1 = plot(MHULL, title="MHULL", color=hullColor, linewidth=thicknessSwitch)
Fi2 = plot(visualSwitch ? SHULL : na, title="SHULL", color=hullColor, linewidth=thicknessSwitch)
alertcondition(ta.crossover(MHULL, SHULL), title="Hull trending up.", message="Hull trending up.")
alertcondition(ta.crossover(SHULL, MHULL), title="Hull trending down.", message="Hull trending down.")

// FILL BAND
fill(Fi1, Fi2, title="Band Filler", color=hullColor)

// BAR COLORING
barcolor(color = candleCol ? (switchColor ? hullColor : na) : na)

//Ichimoku
conversionPeriods = input.int(9, minval=1, title="Conversion Line Length", group = "Ichimoku")
basePeriods = input.int(26, minval=1, title="Base Line Length", group = "Ichimoku")
laggingSpan2Periods = input.int(52, minval=1, title="Leading Span B Length", group = "Ichimoku")
displacement = input.int(26, minval=1, title="Lagging Span", group = "Ichimoku")
donchian(len) => math.avg(ta.lowest(len), ta.highest(len))
conversionLine = donchian(conversionPeriods)
baseLine = donchian(basePeriods)
leadLine1 = math.avg(conversionLine, baseLine)
leadLine2 = donchian(laggingSpan2Periods)
plot(conversionLine, color=color.rgb(60, 165, 250), title="Conversion Line", linewidth = 2)
plot(baseLine, color=#eb2525, title="Base Line", linewidth = 3)
plot(close, offset = -displacement + 1, color=#43A047, title="Lagging Span")
p1 = plot(leadLine1, offset = displacement - 1, color=#A5D6A7,
	 title="Leading Span A")
p2 = plot(leadLine2, offset = displacement - 1, color=#EF9A9A,
	 title="Leading Span B")
plot(leadLine1 > leadLine2 ? leadLine1 : leadLine2, offset = displacement - 1, title = "Kumo Cloud Upper Line", display = display.none) 
plot(leadLine1 < leadLine2 ? leadLine1 : leadLine2, offset = displacement - 1, title = "Kumo Cloud Lower Line", display = display.none)

var bool TouchingUpperPC = false
var bool TouchingLowerPC = false
if high15M[1] == PChighest1 or high15M[2] == PChighest1
    TouchingUpperPC := true
else if low15M[1] == PClowest1 or low15M[2] == PClowest1
    TouchingLowerPC := true
else
    TouchingUpperPC := false
    TouchingLowerPC := false

var bool previousIsRed = false
var bool previousIsGreen = false
var int alarmIsTriggered = 0
if open1H[0] > close1H[0] and open15M[1] > close15M[1] and open15M[0] > close15M[0] and TouchingUpperPC == true
    previousIsRed := true
    previousIsGreen := false
    alarmIsTriggered := alarmIsTriggered + 1
else if open1H[0] < close1H[0] and open15M[1] < close15M[1] and open15M[0] < close15M[0] and TouchingLowerPC == true
    previousIsRed := false
    previousIsGreen := true
    alarmIsTriggered := alarmIsTriggered + 1
else
    previousIsRed := false
    previousIsGreen := false
    alarmIsTriggered := 0

var string Status = ""
if previousIsRed == true and previousIsGreen == false
    log.info("Red")
    Status := "Sell"
    if timeframe.period == "15"
        label.new(bar_index, high, "" + Status, color=color.red, textcolor=color.white, size=size.small)
    if alarmIsTriggered < 3
        alert("SELL")

else if previousIsRed == false and previousIsGreen == true
    log.info("Green")
    Status := "Buy"
    if timeframe.period == "15"
        label.new(bar_index, high, "" + Status, color=color.blue, textcolor=color.white, size=size.small)
    if alarmIsTriggered < 3
        alert("BUY")
